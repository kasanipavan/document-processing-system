-- =============================================
-- Procedure 3: Get Document Types
-- =============================================
CREATE PROCEDURE [dbo].[GetDocumentTypes]
    @IncludeInactive BIT = 0
AS
BEGIN
    SET NOCOUNT ON;
    
    SELECT 
        dt.Id AS DocumentTypeId,
        dt.Category AS DocumentTypeCategory,
        dt.CreatedAt AS DocumentTypeCreatedAt,
        dt.Description AS DocumentTypeDescription,
        dt.FileExtensions AS AllowedFileExtensions,
        dt.IsActive AS DocumentTypeIsActive,
        dt.Keywords AS DocumentTypeKeywords,
        dt.Name AS DocumentTypeName,
        dt.Priority AS DocumentTypePriority,
        dt.ProcessingRules,
        dt.UpdatedAt AS DocumentTypeUpdatedAt
        
    FROM DocumentTypes dt
    WHERE (@IncludeInactive = 1 OR dt.IsActive = 1)
    ORDER BY dt.Priority ASC, dt.Name ASC;
        
    RETURN 0;
END
